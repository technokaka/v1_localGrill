  GNU nano 7.2                                                                           /etc/nginx/sites-available/local_grill                                                                                    server {
    server_name gestion.localgrill-bf.com;  # ton sous-domaine

    location / {
        proxy_pass http://127.0.0.1:3000;   # Next.js écoute sur localhost:3000
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/gestion.localgrill-bf.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/gestion.localgrill-bf.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
server {
    if ($host = gestion.localgrill-bf.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name gestion.localgrill-bf.com;
    return 404; # managed by Certbot


}
